1.	AXI 為什麼write有3個channel，read只有兩個?

	因為slave需要處理寫入的時間，可以先去處理其他事情，等到收到BRESP才算是完成這筆寫入。
	read操作則是slave會回傳data給master，response是合併在read channel裡的，除了有RDATA還有RRESP，拿到資料的同時就知道這筆讀取是成功還是失敗。
	
2.	什麼情況不能out of order?	

	a. 相同ID的txn : 必須跟發送req的順序一致，避免寫入的是舊資料。
	b. 具有相依性的address : 避免讀到舊資料 / 或是寫到舊資料
	c. exclusive access : 當使用AxLock進行atomic操作，這段期間的txn必須照順序執行。

4.	寫印出小於某數的質數?
	
	邏輯 : 假設某數為N，弄一個迴圈次數為根號N，e.g. 數字為23，只要找數字從2到根號23，也就是2~4
	然後把2~4的倍數全部刪掉，剩下沒被刪掉的數字，就是質數。
	
	GPT code:
	#include <stdio.h>
	#include <stdbool.h>
	#include <string.h>

	void findPrimes(int n) {
		// 建立一個布林陣列，初始化為 true
		bool is_prime[n + 1];
		memset(is_prime, true, sizeof(is_prime));

		is_prime[0] = is_prime[1] = false; // 0 和 1 不是質數

		for (int p = 2; p * p <= n; p++) {
			// 如果 is_prime[p] 沒有被劃掉，那它就是質數
			if (is_prime[p] == true) {
				// 劃掉所有 p 的倍數 (從 p*p 開始優化效率)
				for (int i = p * p; i <= n; i += p)
					is_prime[i] = false;
			}
		}

		// 印出結果
		printf("小於等於 %d 的質數有:\n", n);
		for (int p = 2; p <= n; p++) {
			if (is_prime[p]) {
				printf("%d ", p);
			}
		}
		printf("\n");
	}

	int main() {
		int limit = 50;
		findPrimes(limit);
		return 0;
	}

8.	如何驗證一個FIFO
	1. 基礎邊界測試 (Boundary Conditions)
	FIFO Full檢查 : 連續寫入16筆資料，確認full準確拉高，且此時wr_ptr指回起始點
	FIFO Empty檢查: 在有資料的情況下讀取直到清空，確認empty準確拉高，且rd_data符合最後一筆有效值

	2. 資料完整性測試
	寫入一系列數值，透過scoreboard內部memory比對rd_data是否一致。

	3. 異常行為測試
	Write when Full : 在full拉高的情況下再測試wr_en = 1，確認資料不會被overwrite，且full訊號不變。
	Read when empty : 在empty拉高的情況測試rd_en = 1，確認rdptr不會偏移。

	4. 
	Full狀態同時讀寫 : 同時發起wr_en, rd_en。理論上FIFO應維持full，波型應該是短暫拉下來再拉高。
	必須確保寫入新的空位，並讀書正確舊值。

	Empty狀態同時讀寫 : 測試資料是否能在下一個時脈正確呈現

	module counter(
		input clk,
		input rstn,
		input inverse,
		output bit [3:0] result
	);
		logic [3:0] count;		
		assign result = count;
		
		
		always_ff @(posedge clk) begin
			if(!rstn) begin
				count <= 0;
			end
			else begin
				if(inverse==1'b0) begin
					count <= count + 1;
				end
				else begin
					count <= count - 1;
				end
			end
		end
		

	endmodule

10.	寫出驗證上題的seq / seqr / monitor / driver
	//---IF---//
	interface counter_if(input logic clk);
		logic rstn;
		logic inverse;
		logic [3:0] result;
	endinterface
	
	
	//---TXN---//
	class counter_transaction extends uvm_sequence_item;
		`uvm_object_utils(counter_transaction)

		rand logic         inverse;
		logic [3:0]        result; // 這是從 DUT 讀回來的，不需要 rand

		function new(string name = "counter_transaction");
			super.new(name);
		endfunction
	endclass
	
	
	
	//---Seqr---//
	// Sequencer
	typedef uvm_sequencer #(counter_transaction) counter_sequencer;

	// Sequence: 隨機測試 20 次計數
	class counter_base_sequence extends uvm_sequence #(counter_transaction);
		`uvm_object_utils(counter_base_sequence)

		task body();
			repeat(20) begin
				req = counter_transaction::type_id::create("req");
				start_item(req);
				if (!req.randomize()) `uvm_error("SEQ", "Randomization failed")
				finish_item(req);
			end
		endtask
	endclass
	
	
	//---Driver---//
	class counter_driver extends uvm_driver #(counter_transaction);
		`uvm_component_utils(counter_driver)
		
		virtual counter_if vif;

		function new(string name, uvm_component parent);
			super.new(name, parent);
		endfunction

		// 從 config_db 取得虛擬介面
		virtual function void build_phase(uvm_phase phase);
			super.build_phase(phase);
			if (!uvm_config_db#(virtual counter_if)::get(this, "", "vif", vif))
				`uvm_fatal("DRV", "Could not get vif")
		endfunction

		virtual task run_phase(uvm_phase phase);
			// 初始化信號
			vif.inverse <= 0;
			vif.rstn    <= 0;
			#20 vif.rstn <= 1; // 釋放重置

			forever begin
				seq_item_port.get_next_item(req);
				@(posedge vif.clk);
				vif.inverse <= req.inverse; // 將驅動信號上到 Pin
				seq_item_port.item_done();
			end
		endtask
	endclass
	
	
	//---Monitor---//
	class counter_monitor extends uvm_monitor;
		`uvm_component_utils(counter_monitor)

		virtual counter_if vif;
		uvm_analysis_port #(counter_transaction) ap;

		function new(string name, uvm_component parent);
			super.new(name, parent);
			ap = new("ap", this);
		endfunction

		virtual function void build_phase(uvm_phase phase);
			super.build_phase(phase);
			if (!uvm_config_db#(virtual counter_if)::get(this, "", "vif", vif))
				`uvm_fatal("MON", "Could not get vif")
		endfunction

		virtual task run_phase(uvm_phase phase);
			counter_transaction tr;
			forever begin
				@(posedge vif.clk);
				tr = counter_transaction::type_id::create("tr");
				tr.inverse = vif.inverse;
				tr.result  = vif.result; // 取回當前 DUT 的輸出值
				ap.write(tr);            // 廣播給 Scoreboard
			end
		endtask
	endclass
	
	
11.	請設計一個sync fifo
	
	
	module sync_fifo #(
    parameter DATA_WIDTH = 32,
    parameter DEPTH      = 8,
    parameter ADDR_WIDTH = $clog2(DEPTH)
)(
    input  logic                   clk,
    input  logic                   rst_n,
    
    // Write Interface
    input  logic                   wr_en,
    input  logic [DATA_WIDTH-1:0]  wr_data,
    output logic                   full,
    
    // Read Interface
    input  logic                   rd_en,
    output logic [DATA_WIDTH-1:0]  rd_data,
    output logic                   empty,
    
    // Status
    output logic [ADDR_WIDTH:0]    fifo_cnt
);

		// Memory array
		logic [DATA_WIDTH-1:0] fifo_mem [DEPTH];
		
		logic [2:0] wr_ptr; //index = 0-7
		logic [2:0] rd_ptr; //index = 0-7

		// --- Write Logic ---
		always_ff @(posedge clk or negedge rst_n) begin
			if (!rst_n) begin
				wr_ptr <= 0;
			end else if (wr_en && !full) begin
				fifo_mem[wr_ptr] <= wr_data;
				wr_ptr <= wr_ptr + 1;
			end
		end

		// --- Read Logic ---
		always_ff @(posedge clk or negedge rst_n) begin
			if (!rst_n) begin
				rd_ptr  <= 0;
				rd_data <= 0;
			end else if (rd_en && !empty) begin
				rd_data <= fifo_mem[rd_ptr];
				rd_ptr  <= rd_ptr + 1;
			end
		end

		// --- Counter & Status Flags ---
		always_ff @(posedge clk or negedge rst_n) begin
			if (!rst_n) begin
				fifo_cnt <= 0;
			end else begin
				case ({wr_en && !full, rd_en && !empty})
					2'b10: fifo_cnt <= fifo_cnt + 1; // 只寫不讀
					2'b01: fifo_cnt <= fifo_cnt - 1; // 只讀不寫
					default: fifo_cnt <= fifo_cnt;   // 不動或同時讀寫
				endcase
			end
		end

		assign full  = (fifo_cnt == DEPTH);
		assign empty = (fifo_cnt == 0);

	endmodule
	
	

12.	設計一個除頻器 => 1/2 頻率

	想像一個水龍頭，輸入端 clk_in 是每秒滴水一次（頻率 $1\text{Hz}$）。
	這個除頻器的邏輯是：「看到一次滴水，就切換一次開關」。
	第 1 次滴水：開關從「關」變成「開」。
	第 2 次滴水：開關從「開」變成「關」。
	結果是：輸出端 clk_out 完成「一開一關」的完整循環需要 2 次 滴水。
	所以，輸出的頻率就變成了輸入的一半（$0.5\text{Hz}$），這就是 二除頻 (Divide-by-2)。

	module clk_div(
		input wire clk_in, 
		input wire enable,
		input wire reset,
		output reg clk_out
	);
		always_ff @(posedge clk_in)
			if(!reset) clk_out <= 1'b0;
			else if(enable) clk_out <= !clk_out;		
	endmodule
	
	
12-1.設計一個除頻器 => 1/3 頻率
	概念上是使用正緣 & 負緣都增加切換開關的"時間"
	之前是每正緣數一次就切換開關，但現在是數三次才切換開關
	
	module clk_div_3(
		input logic clk_in;
		input logic rst_n,
		output logic clk_out
	);
		logic [1:0] pos_cnt;
		logic [1:0] neg_cnt;
		
		// pos counter
		always_ff @(posedge clk_in or negedge rst_n) begin
			if(!rst_n) pos_cnt <= 2'b0;
			else begin 
				if(pos_cnt == 2'd2) pos_cnt <= 2'd0;
				else pos_cnt <= pos_cnt + 1'b1;
			end
		end	
		
		always_ff @(negedge clk_in or negedge rst_n) begin
			if(!rst_n) neg_cnt <= 2'b0;
			else begin
				if(neg_cnt == 2'd2) neg_cnt <= 2'd0;
				else neg_cnt <= neg_cnt + 1'b1;
			end
		end	
		
		assign clk_out = (pos_cnt != 2) && (neg_cnt != 2);
	endmodule

14.	solve before介紹
	請列出n=0的機率
	rand bit a, m;
	rand bit [1:0] b, n;
	
	constraint c1{a -> b==0;}
	constraint c2{m -> n==0; solve m before n}
	
	1. c1總共的可能性:
	[0, 0]
	[0, 1]
	[0, 2]
	[0, 3]
	[1, 0]
	以上每種可能都是1/5
	n=0的機率為2/5
	
	2. c2的可能性為:
	先檢查m的所有可能性 => 1/2 ，再檢查n
	如果m = 0，此時n可能為{0, 1, 2, 3}，所以每種可能性是1/4   => 所以n在這裡等於0的可能性為1/2 * 1/4 = 1/8
	如果m = 1，n只能為0的機率是100%  => 1/2 * 100% = 1/2
	所以n=0的組合是[0, 0], [1, 0] => 1/8 + 1/2 = 5/8
